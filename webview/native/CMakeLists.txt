cmake_minimum_required(VERSION 3.21)
project(lut_cef LANGUAGES C CXX)
if(APPLE)
  enable_language(OBJCXX)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(USE_SANDBOX OFF CACHE BOOL "Development prototype only" FORCE)
if(NOT CEF_ROOT)
  message(FATAL_ERROR "Set CEF_ROOT to an extracted official CEF binary distribution")
endif()
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" wrapper)
add_library(lut_cef SHARED bridge.cc)
add_executable(lut_cef_helper WIN32 helper.cc)
foreach(target lut_cef lut_cef_helper)
  target_include_directories(${target} PRIVATE "${CEF_ROOT}")
  SET_EXECUTABLE_TARGET_PROPERTIES(${target})
  target_link_libraries(${target} PRIVATE libcef_dll_wrapper ${CEF_STANDARD_LIBS})
endforeach()
if(APPLE)
  enable_language(OBJCXX)
  target_sources(lut_cef PRIVATE cocoa.mm)
else()
  ADD_LOGICAL_TARGET("libcef_lib" "${CEF_LIB_DEBUG}" "${CEF_LIB_RELEASE}")
  target_link_libraries(lut_cef PRIVATE libcef_lib)
  target_link_libraries(lut_cef_helper PRIVATE libcef_lib)
  if(UNIX)
    set_target_properties(lut_cef lut_cef_helper PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
  endif()
endif()
install(TARGETS lut_cef lut_cef_helper RUNTIME DESTINATION . LIBRARY DESTINATION .)
