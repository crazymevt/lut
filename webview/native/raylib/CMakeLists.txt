cmake_minimum_required(VERSION 3.21)
project(lut_raylib_shim C)
add_library(raylib-shim SHARED "${BINDING_ROOT}/src/generated/shim.c")
set_target_properties(raylib-shim PROPERTIES PREFIX "lib" BUILD_WITH_INSTALL_RPATH TRUE)
target_include_directories(raylib-shim PRIVATE "${RAYLIB_ROOT}/include")
if(WIN32)
  target_link_libraries(raylib-shim PRIVATE "${RAYLIB_ROOT}/lib/raylibdll.lib")
else()
  find_library(RAYLIB_LIBRARY raylib PATHS "${RAYLIB_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
  target_link_libraries(raylib-shim PRIVATE "${RAYLIB_LIBRARY}")
  if(APPLE)
    set_target_properties(raylib-shim PROPERTIES INSTALL_RPATH "@loader_path")
  else()
    set_target_properties(raylib-shim PROPERTIES INSTALL_RPATH "$ORIGIN")
  endif()
endif()
install(TARGETS raylib-shim RUNTIME DESTINATION . LIBRARY DESTINATION .)
